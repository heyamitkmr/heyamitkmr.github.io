# 404 — nothing here

The page you're looking for doesn't exist (or has moved). Try the [writing index](/writing/).

<script>
  (function () {
    // Old-style post URLs: /blog|bits|project|product/YYYY/MM/DD/slug.html
    var m = location.pathname.match(/^\/(blog|bits|project|product)\/\d{4}\/\d{1,2}\/\d{1,2}\/(.+?)(\/?)$/i);
    if (!m) return;
    var section = m[1].toLowerCase() === "blog" ? "writing" : m[1].toLowerCase();
    var target = "/" + section + "/" + m[2];
    // Bounce only if the target exists (prevents loops on bad slugs)
    fetch(target, { method: "HEAD" })
      .then(function (r) { if (r.ok) location.replace(target); })
      .catch(function () {});
  })();
</script>
