<script>
  (function () {
    var m = location.pathname.match(/^\/(blog|bits|project|product)\/\d{4}\/\d{1,2}\/\d{1,2}\/(.+?)(\/?)$/i);
    if (!m) return;
    var section = m[1].toLowerCase() === "blog" ? "writing" : m[1].toLowerCase();
    var target = "/" + section + "/" + m[2];
    fetch(target, { method: "HEAD" })
      .then(function (r) { if (r.ok) location.replace(target); })
      .catch(function () {});
  })();
</script>

<script>
  var N5_BITS = [
    {%- for post in site.categories.bits -%}
    {%- assign dq = '"' -%}
    {%- assign img_marker = 'src=' | append: dq -%}
    {%- assign img_parts = post.content | split: img_marker -%}
    {%- assign bit_img = "" -%}
    {%- if img_parts.size > 1 -%}
      {%- assign bit_img = img_parts[1] | split: dq | first -%}
    {%- endif -%}
    {
      "t": {{ post.title | jsonify }},
      "u": "{{ post.url }}",
      "d": "{{ post.date | date: '%b %-d, %Y' }}"{% if bit_img != "" %},
      "i": "{{ bit_img }}"{% endif %}
    }{%- unless forloop.last -%},{%- endunless -%}
    {%- endfor -%}
  ];
  (function () {
    if (!window.N5_BITS || !N5_BITS.length) return;
    var pick = N5_BITS[Math.floor(Math.random() * N5_BITS.length)];
    var link = document.getElementById("random-bit-link");
    link.textContent = pick.t;
    link.href = pick.u;
    document.getElementById("random-bit-date").textContent = " · " + pick.d;
    if (pick.i) {
      var img = document.createElement("img");
      img.src = pick.i;
      img.alt = "";
      img.loading = "lazy";
      img.decoding = "async";
      img.style.maxWidth = "100%";
      img.style.height = "auto";
      document.getElementById("random-bit").appendChild(img);
    }
    document.getElementById("random-bit").hidden = false;
  })();
</script>
